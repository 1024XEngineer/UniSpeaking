@Service
public class FreeChatSessionService implements SessionService {

    // ========================================================
    // 1. 启动自由聊天
    // 前端通过 HTTP 发送 Offer SDP
    // ========================================================

    public StartResult start(StartCommand command) {
        // 获取当前用户和用户画像
        String userId =
            AuthService.getCurrentUserId();
        UserProfile profile =
            ProfileService.getProfile(userId);

        // 获取自由聊天场景配置并校验访问权限
        SceneConfig sceneConfig =
            SceneService.getByType(SceneType.FREE_CHAT);
        SceneService.validateAccess(
            userId,
            sceneConfig
        );
        UsageQuotaService.validate(
            userId,
            SceneType.FREE_CHAT
        );


        // 创建本地业务会话
        String localSessionId =
            SessionIdGenerator.generate();
        FreeChatSceneSession session =
            new FreeChatSceneSession(
                localSessionId,
                userId
            );

        // 生成自由聊天 Prompt
        SessionPrompt prompt =
            FreeChatPromptService.build(
                profile,
                sceneConfig
            );
        session.setSceneId(sceneConfig.getSceneId());
        session.setProviderType(
            sceneConfig.getProviderType()
        );
        session.setPrompt(prompt);
        session.setVoiceId(
            profile.getVoiceId() != null
                ? profile.getVoiceId()
                : sceneConfig.getDefaultVoiceId()
        );

        // 预占用户额度
        UsageQuotaService.reserve(
            userId,
            localSessionId
        );

        // 保存本地会话对象
        SessionStateStore.save(session);

        try {

            // 服务端代理完成 SDP 交换
            String answerSdp =
                RealtimeConnectionService.createAnswerSdp(
                    session.getProviderType(),
                    command.getOfferSdp()
                );

            // 等待模型返回 providerSessionId
            session.setState()Connecting();
            SessionStateStore.save(session);

            // 启动绑定超时检测
            SessionBindingTimeoutService.schedule(
                localSessionId,
                BIND_TIMEOUT
            );

            // 返回前端建连信息
            return StartResult.builder()
                .setLocalSessionId(localSessionId)
                .setAnswerSdp(connection.getAnswerSdp())
                .setSystemPrompt(prompt)
                .setVoiceId(session.getVoiceId())
                .build();

        } catch (Exception exception) {

            session.fail(
                "SESSION_START_FAILED",
                exception.getMessage()
            );

            SessionStateStore.save(session);

            UsageQuotaService.release(
                localSessionId
            );

            throw exception;
        }
    }


    // ========================================================
    // 2. 统一处理会话事件
    //
    // 普通事件：
    // SessionWebSocketHandler → handleEvent()
    //
    // 绑定超时：
    // SessionBindingTimeoutService → handleEvent(BIND_TIMEOUT)
    // ========================================================

    public void handleEvent(
            String localSessionId,
            SessionEvent event) {

        FreeChatSceneSession session =
            SessionStateStore.get(localSessionId);

        switch (event.getType()) {

            // 模型通过 DataChannel 返回 session.created，
            // 前端再通过 WebSocket 上报给后端
            case PROVIDER_SESSION_CREATED -> {
                session.bindProviderSession(
                    event.getProviderSessionId()
                );
                SessionBindingTimeoutService.cancel(
                    localSessionId
                );
                UsageQuotaService.startMetering(
                    localSessionId
                );
                SessionStateStore.save(session);
            }

            // 模型的 Prompt、音色等配置完成
            case PROVIDER_SESSION_UPDATED -> {
                session.activate();
                SessionStateStore.save(session);
            }

            // 用户或 AI 的完整字幕
            case USER_TRANSCRIPT_COMPLETED,
                 AI_TRANSCRIPT_COMPLETED -> {
                FreeChatConversationService.appendMessage(
                    localSessionId,
                    event.toConversationMessage()
                );
            }


            // 用户暂停
            case SESSION_PAUSE -> {
                session.pause();
                SessionStateStore.save(session);
            }


            // 用户恢复
            case SESSION_RESUME -> {
                session.resume();
                SessionStateStore.save(session);
            }


            // 用户打断 AI
            case SESSION_INTERRUPT -> {
                session.recordInterrupt();
                SessionStateStore.save(session);
                // 真正停止模型回答：
                // 前端同时通过 DataChannel 向模型发送打断事件
            }


            // 用户正常结束
            case SESSION_COMPLETE -> {
                session.complete();
                SessionStateStore.save(session);
                UsageQuotaService.settle(
                    localSessionId
                );
                // Conversation Store 保留逐轮消息。
                // 不生成会话摘要，不更新 user_preference.memory_text。
            }


            // 模型、WebRTC 或 WebSocket 异常
            case SESSION_ERROR -> {
                session.fail(
                    event.getErrorCode(),
                    event.getErrorMessage()
                );
                SessionStateStore.save(session);
                UsageQuotaService.settle(
                    localSessionId
                );
            }


            // 规定时间内没有绑定 providerSessionId
            case BIND_TIMEOUT -> {
                if (session.getProviderSessionId() == null) {
                    session.fail(
                        "PROVIDER_BIND_TIMEOUT",
                        "模型会话绑定超时"
                    );
                    SessionStateStore.save(session);
                    UsageQuotaService.settleReservedQuota(
                        localSessionId
                    );
                    SessionWebSocketHandler.sendCloseWebRtc(
                        localSessionId
                    );
                }
            }
        }
    }
}
