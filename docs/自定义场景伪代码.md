public void runCustomScene(
        CreateCustomSceneCommand command) {
    // =====================================================
    // 1. 创建并生成自定义场景
    // =====================================================

    String userId = AuthService.getCurrentUserId();

    CustomScene scene =
        SceneService.create(
            userId,
            command.getSceneInput(),
            command.getLevel()
        );

    CustomSceneContent content =
        CustomSceneGenerationService.generate(
            scene.getSceneId(),
            command.getSceneInput(),
            command.getLevel()
        );

    // 保存到本地 HashMap
    SceneContentService.setLocal(
        scene.getSceneId(),
        content
    );

    // 异步保存到数据库
    SceneContentService.saveAsync(
        scene.getSceneId(),
        content
    );

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.WORD
    );


    // =====================================================
    // 2. 打开学习阶段 WebSocket
    // =====================================================

    CustomSceneLearningService.open(
        userId,
        scene.getSceneId()
    );


    // =====================================================
    // 3. 单词学习
    // 前端负责依次展示多个单词
    // =====================================================

    List<LearningMaterial> words =
        SceneContentService.getLocal(
            scene.getSceneId(),
            CustomStage.WORD
        );

    CustomSceneLearningService.setMaterials(
        scene.getSceneId(),
        words
    );

    // 每次收到播放请求
    AudioStream wordAudio =
        TtsService.getAudio(
            currentWord.getContent()
        );

    CustomSceneLearningService.setAudio(
        scene.getSceneId(),
        wordAudio
    );

    // 用户重复点击时，再次调用 TTS
    // TtsService.getAudio(currentWord.getContent())

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.PHRASE
    );


    // =====================================================
    // 4. 词组学习
    // =====================================================

    List<LearningMaterial> phrases =
        SceneContentService.getLocal(
            scene.getSceneId(),
            CustomStage.PHRASE
        );

    CustomSceneLearningService.setMaterials(
        scene.getSceneId(),
        phrases
    );

    // 每次收到播放请求
    AudioStream phraseAudio =
        TtsService.getAudio(
            currentPhrase.getContent()
        );

    CustomSceneLearningService.setAudio(
        scene.getSceneId(),
        phraseAudio
    );

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.SENTENCE
    );


    // =====================================================
    // 5. 句子跟读
    // =====================================================

    List<LearningMaterial> sentences =
        SceneContentService.getLocal(
            scene.getSceneId(),
            CustomStage.SENTENCE
        );

    CustomSceneLearningService.setMaterials(
        scene.getSceneId(),
        sentences
    );

    for (LearningMaterial sentence : sentences) {

        boolean passed = false;

        while (!passed) {

            // 播放标准读音
            AudioStream standardAudio =
                TtsService.getAudio(
                    sentence.getContent()
                );

            CustomSceneLearningService.setAudio(
                scene.getSceneId(),
                standardAudio
            );

            // 开始接收用户音频
            String assessmentId =
                PronunciationService.start(
                    sentence.getContent()
                );

            // WebSocket 每收到一个音频分片
            PronunciationService.setAudio(
                assessmentId,
                audioChunk
            );

            // 音频结束后获取音素评分
            PronunciationResult score =
                PronunciationService.getResult(
                    assessmentId
                );

            passed =
                score.getTotalScore() >= 80;

            // 更新本地分数
            SceneContentService.setSentenceScore(
                scene.getSceneId(),
                sentence.getMaterialId(),
                score,
                passed
            );

            // 异步保存数据库
            SceneContentService.saveSentenceScoreAsync(
                scene.getSceneId(),
                sentence.getMaterialId(),
                score,
                passed
            );

            // 将评分返回学习 WebSocket
            CustomSceneLearningService.setSentenceScore(
                scene.getSceneId(),
                sentence.getMaterialId(),
                score
            );

            // 低于80分继续当前循环
            // 高于80分进入下一个句子
        }
    }


    // =====================================================
    // 6. 学习完成，关闭学习 WebSocket
    // =====================================================

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.DIALOGUE_READY
    );

    CustomSceneLearningService.close(
        scene.getSceneId()
    );


    // =====================================================
    // 7. 启动 Realtime 场景会话
    // 复用自由聊天的建连流程
    // =====================================================

    DialoguePromptData promptData =
        SceneContentService.getDialoguePrompt(
            scene.getSceneId()
        );

    SessionPrompt prompt =
        CustomScenePromptService.getPrompt(
            userId,
            scene,
            promptData
        );

    StartResult startResult =
        CustomSessionService.start(
            scene.getSceneId(),
            prompt,
            offerSdp
        );

    String localSessionId =
        startResult.getLocalSessionId();

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.DIALOGUE
    );


    // =====================================================
    // 8. Realtime 对话过程
    // =====================================================

    while (!CustomSessionService.isCompleted(
            localSessionId)) {

        String utteranceId =
            CustomSceneEvaluationService.start(
                localSessionId
            );

        // WebSocket 每收到一个用户音频分片
        CustomSceneEvaluationService.setAudio(
            localSessionId,
            utteranceId,
            audioChunk
        );

        // 用户音频结束
        CustomSceneEvaluationService.finishAudio(
            localSessionId,
            utteranceId
        );

        // 接收前端同步的用户字幕
        SessionConversationService.setMessage(
            localSessionId,
            utteranceId,
            Role.USER,
            userTranscript
        );

        SessionConversationService.saveMessageAsync(
            localSessionId,
            utteranceId,
            Role.USER,
            userTranscript
        );

        // 字幕交给评分服务
        CustomSceneEvaluationService.setTranscript(
            localSessionId,
            utteranceId,
            userTranscript
        );

        // 获取五维评分和总分
        TurnEvaluationResult turnScore =
            CustomSceneEvaluationService.getTurnResult(
                localSessionId,
                utteranceId
            );

        EvaluationRecordService.setTurnResult(
            localSessionId,
            utteranceId,
            turnScore
        );

        EvaluationRecordService.saveTurnResultAsync(
            turnScore
        );

        // 异步生成推荐表达和错误纠正
        RecommendedExpression recommendation =
            LlmService.getRecommendedExpressionAsync(
                localSessionId,
                userTranscript
            );

        EvaluationRecordService.setRecommendation(
            localSessionId,
            utteranceId,
            recommendation
        );

        EvaluationRecordService.saveRecommendationAsync(
            localSessionId,
            utteranceId,
            recommendation
        );

        // 保存 AI 字幕
        SessionConversationService.setMessage(
            localSessionId,
            null,
            Role.ASSISTANT,
            aiTranscript
        );

        SessionConversationService.saveMessageAsync(
            localSessionId,
            null,
            Role.ASSISTANT,
            aiTranscript
        );
    }


    // =====================================================
    // 9. AI 正常结束，生成最终报告
    // =====================================================

    List<TurnEvaluationResult> turnResults =
        EvaluationRecordService.getTurnResults(
            localSessionId
        );

    List<ConversationMessage> messages =
        SessionConversationService.getMessages(
            localSessionId
        );

    CustomSceneEvaluationReport report =
        CustomSceneEvaluationService.getFinalReport(
            localSessionId,
            turnResults,
            messages
        );

    EvaluationRecordService.setReport(
        localSessionId,
        report
    );

    EvaluationRecordService.saveReportAsync(
        report
    );

    SessionConversationService.saveConversationAsync(
        localSessionId,
        messages
    );

    SceneService.setCompletedAsync(
        scene.getSceneId(),
        localSessionId
    );

    SceneProgressService.set(
        scene.getSceneId(),
        CustomStage.REPORT
    );


    // =====================================================
    // 10. 关闭连接并清理本地数据
    // =====================================================

    CustomSessionService.close(
        localSessionId
    );

    SessionConversationService.clearLocal(
        localSessionId
    );

    EvaluationRecordService.clearLocal(
        localSessionId
    );

    SceneContentService.expireLocal(
        scene.getSceneId()
    );
}
